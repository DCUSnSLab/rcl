cmake_minimum_required(VERSION 3.16)
project(rcl_executor)
find_package(ament_cmake REQUIRED)
find_package(rcl REQUIRED)
find_package(Threads REQUIRED)
add_library(${PROJECT_NAME} SHARED src/executor.cpp)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
target_compile_definitions(${PROJECT_NAME} PRIVATE RCL_EXECUTOR_BUILDING_LIBRARY)
set_target_properties(${PROJECT_NAME} PROPERTIES CXX_VISIBILITY_PRESET hidden
  VISIBILITY_INLINES_HIDDEN ON)
target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(${PROJECT_NAME} PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_link_libraries(${PROJECT_NAME} PUBLIC rcl::rcl PRIVATE Threads::Threads)
install(TARGETS ${PROJECT_NAME} EXPORT ${PROJECT_NAME} LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_dependencies(rcl)
if(BUILD_TESTING)
  add_executable(test_c_api test/test_c_api.c)
  target_link_libraries(test_c_api ${PROJECT_NAME})
  add_test(NAME test_c_api COMMAND test_c_api)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_executor test/test_executor.cpp)
  target_link_libraries(test_executor ${PROJECT_NAME} Threads::Threads)
endif()
ament_package()
