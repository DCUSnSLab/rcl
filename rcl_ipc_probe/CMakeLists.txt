cmake_minimum_required(VERSION 3.16)
project(rcl_ipc_probe)
find_package(ament_cmake REQUIRED)
find_package(rcl REQUIRED)
find_package(rcl_executor REQUIRED)
find_package(sensor_msgs REQUIRED)
option(WP4_IPC "Build the owned C message experiment" OFF)
add_library(${PROJECT_NAME} SHARED probe.cpp)
target_compile_features(${PROJECT_NAME} PRIVATE cxx_std_17)
target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(${PROJECT_NAME} rcl::rcl rcl_executor::rcl_executor ${sensor_msgs_TARGETS})
if(WP4_IPC)
  target_compile_definitions(${PROJECT_NAME} PRIVATE WP4_IPC)
endif()
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
install(PROGRAMS bench.py DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING AND WP4_IPC)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_ipc test_ipc.cpp)
  target_include_directories(test_ipc PRIVATE ../rcl/src)
  target_link_libraries(test_ipc rcl::rcl rcl_executor::rcl_executor ${sensor_msgs_TARGETS})
endif()
ament_package()
